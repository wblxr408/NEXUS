cmake_minimum_required(VERSION 3.16)
project(nexus_rviz_render_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_executable(rviz_render_probe main.cpp)
ament_target_dependencies(rviz_render_probe rclcpp rviz_common rviz_rendering)
target_link_libraries(rviz_render_probe Qt5::Widgets)
